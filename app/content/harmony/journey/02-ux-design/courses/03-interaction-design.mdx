## 详细内容

### 手势操作

```typescript
@Entry
@Component
struct GestureExample {
  @State scale: number = 1.0;

  build() {
    Column() {
      Text('手势操作')
        .fontSize(30)
        .scale({ x: this.scale, y: this.scale })
        .gesture(
          PinchGesture({ fingers: 2 })
            .onActionStart(() => {
              console.log('手势开始');
            })
            .onActionUpdate((event: GestureEvent) => {
              this.scale = event.scale;
            })
            .onActionEnd(() => {
              console.log('手势结束');
            })
        )
    }
    .width('100%')
    .height('100%')
  }
}
```
